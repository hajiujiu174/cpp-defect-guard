cmake_minimum_required(VERSION 3.29)
project(RepairComparisonDemo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
enable_testing()
add_executable(comparison_demo main.cpp zero.cpp one.cpp helpers.cpp)
add_test(NAME "zero [sample]" COMMAND comparison_demo zero)
add_test(NAME one.sample COMMAND comparison_demo one)
add_test(NAME idle COMMAND comparison_demo idle)
add_test(NAME ready COMMAND "${CMAKE_COMMAND}" -E true)
add_test(NAME cleanup COMMAND "${CMAKE_COMMAND}" -E true)
set_tests_properties(ready PROPERTIES FIXTURES_SETUP environment)
set_tests_properties(cleanup PROPERTIES FIXTURES_CLEANUP environment)
set_tests_properties(one.sample PROPERTIES FIXTURES_REQUIRED environment)
